361:     static func clampedCornerRadii(size: CGSize, cornerRadii: CornerRadii) -> CornerRadii {
362:         let size = CGSize(width: max(0.0, size.width), height: max(0.0, size.height))
363:         var cornerRadii = CornerRadii(
364:             topLeft: max(0.0, cornerRadii.topLeft),
365:             topRight: max(0.0, cornerRadii.topRight),
366:             bottomLeft: max(0.0, cornerRadii.bottomLeft),
367:             bottomRight: max(0.0, cornerRadii.bottomRight)
368:         )
369: 
370:         func scaleFor(edgeLength: CGFloat, _ lhs: CGFloat, _ rhs: CGFloat) -> CGFloat {
371:             let sum = lhs + rhs
372:             if sum <= edgeLength || sum.isZero {
373:                 return 1.0
374:             }
375:             return edgeLength / sum
376:         }
377: 
378:         let scale = min(
379:             1.0,
380:             scaleFor(edgeLength: size.width, cornerRadii.topLeft, cornerRadii.topRight),
381:             scaleFor(edgeLength: size.width, cornerRadii.bottomLeft, cornerRadii.bottomRight),
382:             scaleFor(edgeLength: size.height, cornerRadii.topLeft, cornerRadii.bottomLeft),
383:             scaleFor(edgeLength: size.height, cornerRadii.topRight, cornerRadii.bottomRight)
384:         )
385: 
386:         if scale < 1.0 {
387:             cornerRadii = CornerRadii(
388:                 topLeft: cornerRadii.topLeft * scale,
389:                 topRight: cornerRadii.topRight * scale,
390:                 bottomLeft: cornerRadii.bottomLeft * scale,
391:                 bottomRight: cornerRadii.bottomRight * scale
392:             )
393:         }
394: 
395:         return cornerRadii
396:     }

399:         let cornerRadii = self.clampedCornerRadii(size: rect.size, cornerRadii: cornerRadii)

402:         func addCorner(tangent1End: CGPoint, tangent2End: CGPoint, radius: CGFloat) {
403:             if radius > CGFloat.ulpOfOne {
404:                 path.addArc(tangent1End: tangent1End, tangent2End: tangent2End, radius: radius)
405:             } else {
406:                 path.addLine(to: tangent1End)
407:                 path.addLine(to: tangent2End)
408:             }
409:         }
410: 
411:         path.move(to: CGPoint(x: rect.minX, y: rect.minY + cornerRadii.topLeft))
412:         addCorner(
413:             tangent1End: CGPoint(x: rect.minX, y: rect.minY),
414:             tangent2End: CGPoint(x: rect.minX + cornerRadii.topLeft, y: rect.minY),

416:         )
417:         path.addLine(to: CGPoint(x: rect.maxX - cornerRadii.topRight, y: rect.minY))
418:         addCorner(
419:             tangent1End: CGPoint(x: rect.maxX, y: rect.minY),
420:             tangent2End: CGPoint(x: rect.maxX, y: rect.minY + cornerRadii.topRight),
421:             radius: cornerRadii.topRight
422:         )
423:         path.addLine(to: CGPoint(x: rect.maxX, y: rect.maxY - cornerRadii.bottomRight))
424:         addCorner(
425:             tangent1End: CGPoint(x: rect.maxX, y: rect.maxY),
426:             tangent2End: CGPoint(x: rect.maxX - cornerRadii.bottomRight, y: rect.maxY),
427:             radius: cornerRadii.bottomRight

433:             radius: cornerRadii.bottomLeft

583:     /// Puts a new effect view in place of the native one, with the same effect, shape and content (`contentView` is then
584:     /// the new one's). UIKit's press on interactive glass (`_UIFlexInteraction`) belongs to the effect view it started on,
585:     /// and a glass it is pressing is not fused with its neighbours in a glass container: it keeps its own edge inside
586:     /// their merged shape (probed on iOS 27; removing the interaction leaves the press's lift and tint behind, a new effect
587:     /// view is clean). The press ends with the old view; a caller that wants it continues it. Does nothing for the custom
588:     /// glass.
589:     public func renewNativeView() {
590:         guard let nativeParamsView = self.nativeParamsView, let previous = self.nativeView, let shape = self.nativeViewClippingContext?.shape else {

594:         nativeView.overrideUserInterfaceStyle = previous.overrideUserInterfaceStyle
