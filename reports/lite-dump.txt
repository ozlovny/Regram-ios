75: /// The duration and curve of a move animated with `animation`, as `CALayer.animate(from:to:keyPath:duration:delay:
76: /// curve:...)` draws it (`CALayer.makeAnimation` in Display): the part of the move done at a fraction of the duration.
77: /// `.spring` is a real spring only at Display's two native durations and a bezier otherwise.
78: private func lensMoveCurve(_ animation: ComponentTransition.Animation) -> (duration: Double, progress: (Double) -> CGFloat)? {
79:     guard case let .curve(duration, curve) = animation, duration > 0.0 else {
80:         return nil

90:             return (duration, { springAnimationValueAt(spring, CGFloat($0)) })
91:         } else {
